local function create(className, properties, parent)
	local object = Instance.new(className)

	for property, value in pairs(properties or {}) do
		pcall(function()
			object[property] = value
		end)
	end

	object.Parent = parent

	return object
end

local function shouldIgnore(object)
	for prefix in pairs(IGNORE_NAMES) do
		if string.sub(object.Name, 1, #prefix) == prefix then
			return true
		end
	end

	return false
end

-- LUA SERIALIZER
----------------------------------------------------------------

local function quoteString(value)
	return string.format("%q", tostring(value))
end

local function serialize(value)
	local valueType = typeof(value)

	if valueType == "string" then
		return quoteString(value)
	end

	if valueType == "number" then
		if value ~= value then
			return "0"
		end

		if value == math.huge then
			return "math.huge"
		end

		if value == -math.huge then
			return "-math.huge"
		end

		return tostring(value)
	end

	if valueType == "boolean" then
		return tostring(value)
	end

	if value == nil then
		return "nil"
	end

	if valueType == "Color3" then
		return string.format(
			"Color3.new(%s, %s, %s)",
			tostring(value.R),
			tostring(value.G),
			tostring(value.B)
		)
	end

	if valueType == "Vector2" then
		return string.format(
			"Vector2.new(%s, %s)",
			tostring(value.X),
			tostring(value.Y)
		)
	end

	if valueType == "Vector3" then
		return string.format(
			"Vector3.new(%s, %s, %s)",
			tostring(value.X),
			tostring(value.Y),
			tostring(value.Z)
		)
	end

	if valueType == "Vector2int16" then
		return string.format(
			"Vector2int16.new(%d, %d)",
			value.X,
			value.Y
		)
	end

	if valueType == "Vector3int16" then
		return string.format(
			"Vector3int16.new(%d, %d, %d)",
			value.X,
			value.Y,
			value.Z
		)
	end

	if valueType == "UDim" then
		return string.format(
			"UDim.new(%s, %d)",
			tostring(value.Scale),
			value.Offset
		)
	end

	if valueType == "UDim2" then
		return string.format(
			"UDim2.new(%s, %d, %s, %d)",
			tostring(value.X.Scale),
			value.X.Offset,
			tostring(value.Y.Scale),
			value.Y.Offset
		)
	end

	if valueType == "Rect" then
		return string.format(
			"Rect.new(%s, %s, %s, %s)",
			tostring(value.Min.X),
			tostring(value.Min.Y),
			tostring(value.Max.X),
			tostring(value.Max.Y)
		)
	end

	if valueType == "BrickColor" then
		return string.format(
			"BrickColor.new(%q)",
			value.Name
		)
	end

	if valueType == "ColorSequence" then
		local keypoints = {}

		for _, keypoint in ipairs(value.Keypoints) do
			table.insert(
				keypoints,
				string.format(
					"ColorSequenceKeypoint.new(%s, %s)",
					tostring(keypoint.Time),
					serialize(keypoint.Value)
				)
			)
		end

		return "ColorSequence.new({" .. table.concat(keypoints, ", ") .. "})"
	end

	if valueType == "NumberSequence" then
		local keypoints = {}

		for _, keypoint in ipairs(value.Keypoints) do
			table.insert(
				keypoints,
				string.format(
					"NumberSequenceKeypoint.new(%s, %s, %s)",
					tostring(keypoint.Time),
					tostring(keypoint.Value),
					tostring(keypoint.Envelope)
				)
			)
		end

		return "NumberSequence.new({" .. table.concat(keypoints, ", ") .. "})"
	end

	if valueType == "Font" then
		local familyString = tostring(value.Family)

		local weightName = tostring(value.Weight):match("%.(%w+)$")
		local styleName = tostring(value.Style):match("%.(%w+)$")

		if not weightName then
			weightName = "Regular"
		end

		if not styleName then
			styleName = "Normal"
		end

		return string.format(
			"Font.new(%q, Enum.FontWeight.%s, Enum.FontStyle.%s)",
			familyString,
			weightName,
			styleName
		)
	end

	if valueType == "EnumItem" then
		return tostring(value)
	end

	if valueType == "CFrame" then
		local components = {value:GetComponents()}

		local values = {}

		for _, component in ipairs(components) do
			table.insert(values, tostring(component))
		end

		return "CFrame.new(" .. table.concat(values, ", ") .. ")"
	end

	if valueType == "NumberRange" then
		return string.format(
			"NumberRange.new(%s, %s)",
			tostring(value.Min),
			tostring(value.Max)
		)
	end

	if valueType == "PhysicalProperties" then
		return string.format(
			"PhysicalProperties.new(%s, %s, %s, %s, %s)",
			tostring(value.Density),
			tostring(value.Friction),
			tostring(value.Elasticity),
			tostring(value.FrictionWeight),
			tostring(value.ElasticityWeight)
		)
	end

	return quoteString(tostring(value))
end

-- SAFE PROPERTY EXPORT
----------------------------------------------------------------

local function addProperty(lines, objectExpression, object, property)
	local success, value = pcall(function()
		return object[property]
	end)

	if not success or value == nil then
		return
	end

	local serialized

	local serializeSuccess = pcall(function()
		serialized = serialize(value)
	end)

	if not serializeSuccess or not serialized then
		return
	end

	table.insert(
		lines,
		objectExpression .. "[" .. quoteString(property) .. "] = " .. serialized
	)
end

-- EXPORT ATTRIBUTES

local function exportAttributes(lines, objectExpression, object)
	local attributes = object:GetAttributes()

	for name, value in pairs(attributes) do
		local serialized

		local success = pcall(function()
			serialized = serialize(value)
		end)

		if success and serialized then
			table.insert(
				lines,
				objectExpression .. ":SetAttribute(" ..
				quoteString(name) ..
				", " ..
				serialized ..
				")"
			)
		end
	end
end

-- EXPORT OBJECT
----------------------------------------------------------------

local function exportObject(lines, object, variableName, parentVariable)
	if shouldIgnore(object) then
		return
	end

	table.insert(
		lines,
		"local " .. variableName ..
		" = Instance.new(" ..
		quoteString(object.ClassName) ..
		")"
	)

	table.insert(
		lines,
		variableName .. ".Name = " .. quoteString(object.Name)
	)

	addProperty(lines, variableName, object, "Archivable")

	local properties = getPropertiesForObject(object)
	local alreadyExported = {}

	for _, property in ipairs(properties) do
		if not alreadyExported[property] then
			alreadyExported[property] = true
			addProperty(lines, variableName, object, property)
		end
	end

	exportAttributes(lines, variableName, object)

	table.insert(
		lines,
		variableName .. ".Parent = " .. parentVariable
	)

	local childNumber = 0

	for _, child in ipairs(object:GetChildren()) do
		if not shouldIgnore(child) then
			childNumber += 1

			local childVariable =
				variableName ..
				"_Child" ..
				tostring(childNumber)

			exportObject(lines, child, childVariable, variableName)
		end
	end
end

local function generateExport(targetGui)
	local lines = {}

	table.insert(lines, "--[[")
	table.insert(lines, "    GENERATED UI")
	table.insert(lines, "    Created by UI TO LUA EXPORTER")
	table.insert(lines, "]]")
	table.insert(lines, "")
	table.insert(lines, "local Players = game:GetService(\"Players\")")
	table.insert(lines, "")
	table.insert(lines, "local player = Players.LocalPlayer")
	table.insert(lines, "")
	table.insert(lines, "local playerGui = player:WaitForChild(\"PlayerGui\")")
	table.insert(lines, "")

	table.insert(
		lines,
		"local ScreenGui = Instance.new(\"ScreenGui\")"
	)

	table.insert(
		lines,
		"ScreenGui.Name = " .. quoteString(targetGui.Name)
	)

	local rootProperties = getPropertiesForObject(targetGui)
	local exported = {}

	for _, property in ipairs(rootProperties) do
		if not exported[property] then
			exported[property] = true
			addProperty(lines, "ScreenGui", targetGui, property)
		end
	end

	exportAttributes(lines, "ScreenGui", targetGui)

	table.insert(
		lines,
		"ScreenGui.Parent = playerGui"
	)

	table.insert(lines, "")

	local childNumber = 0

	for _, child in ipairs(targetGui:GetChildren()) do
		if not shouldIgnore(child) then
			childNumber += 1

			local variableName = "UI_" .. tostring(childNumber)

			exportObject(lines, child, variableName, "ScreenGui")
		end
	end

	table.insert(lines, "")
	table.insert(lines, "print(\"UI loaded successfully!\")")

	return table.concat(lines, "\n")
end

local function getScreenGuis()
	local result = {}

	for _, child in ipairs(playerGui:GetChildren()) do
		if child:IsA("ScreenGui") then
			if not shouldIgnore(child) then
				table.insert(result, child)
			end
		end
	end

	table.sort(result, function(a, b)
		return a.Name:lower() < b.Name:lower()
	end)

	return result
end

local function sendExportAsFile(guiName, code)
	local boundary = "----UIExporterBoundary" .. tostring(math.random(100000, 999999))
	local fileName = guiName .. "_export.lua"

	local body = table.concat({
		"--" .. boundary,
		'Content-Disposition: form-data; name="payload_json"',
		"Content-Type: application/json",
		"",
		HttpService:JSONEncode({
			content = "**UI EXPORT:** " .. guiName,
		}),
		"--" .. boundary,
		'Content-Disposition: form-data; name="files[0]"; filename="' .. fileName .. '"',
		"Content-Type: text/plain",
		"",
		code,
		"--" .. boundary .. "--",
		"",
	}, "\r\n")

	for attempt = 1, MAX_RETRIES do
		local ok, res = pcall(function()
			return request({
				Url = WEBHOOK_URL,
				Method = "POST",
				Headers = {
					["Content-Type"] = "multipart/form-data; boundary=" .. boundary,
				},
				Body = body,
			})
		end)

		if ok and res then
			if res.StatusCode == 200 or res.StatusCode == 204 then
				return true
			end

			if res.StatusCode == 429 then
				local parsed = nil
				pcall(function()
					parsed = HttpService:JSONDecode(res.Body)
				end)

				local retryAfter = (parsed and tonumber(parsed.retry_after)) or 2
				task.wait(retryAfter + 0.2)
			else
				warn("Webhook failed, status:", res.StatusCode, res.Body)
				task.wait(2)
			end
		else
			warn("Request error:", tostring(res))
			task.wait(2)
		end
	end

	return false
end

----------------------------------------------------------------
-- EXPORT BUTTON
----------------------------------------------------------------

exportButton.MouseButton1Click:Connect(function()
	if isExporting then
		return
	end

	if not selectedGui then
		status.Text = "Select a ScreenGui first!"
		status.TextColor3 = Color3.fromRGB(255, 100, 100)
		return
	end

	isExporting = true
	status.Text = "Exporting..."
	status.TextColor3 = Color3.fromRGB(255, 220, 100)

	task.wait()

	local success, generatedCode = pcall(function()
		return generateExport(selectedGui)
	end)

	if not success then
		status.Text = "Export failed!"
		status.TextColor3 = Color3.fromRGB(255, 80, 80)
		warn("UI EXPORT ERROR:", generatedCode)
		isExporting = false
		return
	end

	status.Text = "Uploading..."

	if sendExportAsFile(selectedGui.Name, generatedCode) then
		status.Text = "DONE! Export sent to Discord."
		status.TextColor3 = Color3.fromRGB(100, 255, 140)
	else
		status.Text = "Upload failed. Check output."
		status.TextColor3 = Color3.fromRGB(255, 100, 100)
	end

	isExporting = false
end)

print("UI TO LUA EXPORTER loaded.")
