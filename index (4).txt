    function getActiveTab() {

    function switchTab(id) {
      if (activeTabId === id) return;
      if (cm && activeTabId != null) {
        var prev = getActiveTab();
        if (prev) {
          prev.content = cm.getValue();
          prev.cursor = cm.getCursor();
        }
      }
      activeTabId = id;
      var tab = getActiveTab();
      if (!tab) return;
      cm.setValue(tab.content);
      cm.setCursor(tab.cursor || { line: 0, ch: 0 });
      cm.focus();
      updateLangLabel(tab);
      renderTabs();
      clearOutput();
    }

    function closeTab(id, e) {
      if (e) e.stopPropagation();
      if (tabs.length <= 1) return;
      var idx = tabs.findIndex(function(t) { return t.id === id; });
      if (idx === -1) return;
      tabs.splice(idx, 1);
      if (activeTabId === id) {
        var newActive = tabs[Math.max(0, idx - 1)];
        activeTabId = null;
        switchTab(newActive.id);
      } else {
        renderTabs();
      }
    }

    function showHints() {
      var cursor = cm.getCursor();
      var token = cm.getTokenAt(cursor);
      var line = cm.getLine(cursor.line);
      var start = token.start;
      var end = cursor.ch;
      var word = line.slice(start, end).toLowerCase();
      var context = "";
      var before = line.slice(0, start).trimEnd();
      var m = before.match(/([\w]+)\s*[\.:]$/);
      if (m) context = m[1].toLowerCase();

      var list = [];
      ROBLOX_API.forEach(function(item) {
        var n = item.name.toLowerCase();
        if (context) {
          if (n.startsWith(context + ".") || n.startsWith(context + ":")) {
            var short = item.name.slice(context.length + 1);
            if (!word || short.toLowerCase().startsWith(word)) {
              list.push({ text: short, displayText: item.name, description: item.desc });
            }
          }
        } else if (!word || n.startsWith(word) || n.indexOf("." + word) !== -1) {
          list.push({ text: item.name, displayText: item.name, description: item.desc });
        }
      });
      if (!context && word) {
        LUAU_KEYWORDS.forEach(function(k) {
          if (k.startsWith(word) && !list.some(function(x) { return x.text === k; })) {
            list.push({ text: k, displayText: k, description: "Luau keyword" });
          }
        });
      }
      if (!list.length) return;

      CodeMirror.showHint(cm, function() {
        return {
          list: list.map(function(item) {
            return {
              text: item.text,
              displayText: item.displayText,
              render: function(el) {
                el.innerHTML = '<span style="color:#4fc1ff">' + escapeHtml(item.displayText) + '</span>';
                if (item.description) {
                  var d = document.createElement("div");
                  d.style.cssText = "font-size:11px;color:#888;margin-top:1px;";
                  d.textContent = item.description;
                  el.appendChild(d);
                }
              }
            };
          }),
          from: CodeMirror.Pos(cursor.line, start),
          to: CodeMirror.Pos(cursor.line, end)
        };
      }, { completeSingle: false });
    }

    function runCode() {
      var tab = getActiveTab();
      if (!tab || !cm) return;
      var code = cm.getValue();
      tab.content = code;
      document.getElementById("side-panel").classList.add("visible");
      closeMenu();

      var out = document.getElementById("output");
      var prev = document.getElementById("preview");

      if (isProbablyHtml(code) || tab.language === "html") {
        document.getElementById("panel-title").textContent = "Preview";
        out.hidden = true;
        prev.hidden = false;
        prev.srcdoc = code;
        return;
      }

      document.getElementById("panel-title").textContent = "Output";
      out.hidden = false;
      prev.hidden = true;
      out.textContent = "";

      var logs = [];
      var lines = code.split("\n");
      var ran = false;
      for (var i = 0; i < lines.length; i++) {
        var trimmed = lines[i].trim();
        if (trimmed.startsWith("--")) continue;
        var pm = trimmed.match(/^print\s*\((.*)\)\s*$/);
        if (pm) {
          ran = true;
          logs.push(splitArgs(pm[1]).map(function(p) { return formatValue(evalSimple(p.trim())); }).join("\t"));
        }
        var wm = trimmed.match(/^warn\s*\((.*)\)\s*$/);
        if (wm) {
          ran = true;
          logs.push("[WARN] " + splitArgs(wm[1]).map(function(p) { return formatValue(evalSimple(p.trim())); }).join("\t"));
        }
      }
      if (!ran) {
        logs.push("(Demo runs top-level print/warn only.)");
        logs.push("Full Luau: https://play.luau.org");
      }
      out.innerHTML = logs.map(function(l) {
        return l.startsWith("[WARN]") ? '<span class="err">' + escapeHtml(l) + '</span>' : escapeHtml(l);
      }).join("\n");
    }

    function splitArgs(str) {
      var args = [], current = "", inStr = false, quote = "";
      for (var i = 0; i < str.length; i++) {
        var c = str[i];
        if (inStr) {
          current += c;
          if (c === quote && str[i - 1] !== "\\") inStr = false;
        } else if (c === '"' || c === "'") {
          inStr = true; quote = c; current += c;
        } else if (c === ",") {
          args.push(current); current = "";
        } else current += c;
      }
      if (current.trim()) args.push(current);
      return args;
    }

    function evalSimple(expr) {
      expr = expr.trim();
      if ((expr.startsWith('"') && expr.endsWith('"')) || (expr.startsWith("'") && expr.endsWith("'")))
        return expr.slice(1, -1);
      if (expr === "true") return true;
      if (expr === "false") return false;
      if (expr === "nil") return null;
      var num = Number(expr);
      if (!Number.isNaN(num) && expr !== "") return num;
      return expr;
    }

    function formatValue(v) {
      if (v === null || v === undefined) return "nil";
      return String(v);
    }

    // Safe storage: localStorage may be blocked in sandboxed iframes / some previews
    var _memStore = {};
    function storageGet(key) {
      try {
        if (typeof localStorage !== "undefined") {
          var v = localStorage.getItem(key);
          if (v != null) return v;
        }
      } catch (e) { /* sandboxed */ }
      return _memStore[key] || "";
    }
    function storageSet(key, val) {
      _memStore[key] = val;
      try {
        if (typeof localStorage !== "undefined") localStorage.setItem(key, val);
      } catch (e) { /* sandboxed — kept in memory for this session */ }
    }
    function getGroqKey() {
      return storageGet("groq_api_key") || "";
    }
    function saveGroqKey(key) {
      storageSet("groq_api_key", key.trim());
    }

    function wantsCodeEdit(userText) {
      var t = userText.toLowerCase();
      var triggers = [
        "edit", "fix", "change", "rewrite", "update", "modify", "replace",
        "add to", "add a", "insert", "make the", "make it", "write the", "write a",
        "write me", "create a", "create the", "implement", "refactor", "improve",
        "apply", "put this", "use this", "set the code", "new script", "full script",
        "complete script", "whole script", "entire script", "code for", "script that",
        "script to", "can you make", "can you write", "can you add", "can you change",
        "can you fix", "please make", "please write", "please add", "please change",
        "please fix", "generate", "give me the code", "give me a script"
      ];
      for (var i = 0; i < triggers.length; i++) {
        if (t.indexOf(triggers[i]) !== -1) return true;
      }
      return false;
    }

    function extractCodeBlock(text) {
      if (!text) return null;
      // ```lua ... ``` or ``` ... ```
      var m = text.match(/```(?:lua|luau|javascript|html|js)?\s*\r?\n([\s\S]*?)```/i);
      if (m) return m[1].replace(/\s+$/, "");
      // ```code``` same line
      m = text.match(/```(?:lua|luau)?\s*([\s\S]*?)```/i);
      if (m) {
        var body = m[1].replace(/^\s*\n/, "").replace(/\s+$/, "");
        if (body.length > 10) return body;
      }
      return null;
    }

    function formatAiHtml(text) {
      var esc = escapeHtml(text);
      esc = esc.replace(/```(?:lua|luau|javascript|html|js)?\s*\n([\s\S]*?)```/gi, function(_, code) {
        return "<pre><code>" + code + "</code></pre>";
      });
      esc = esc.replace(/`([^`]+)`/g, "<code>$1</code>");
      esc = esc.replace(/\n/g, "<br>");
      return esc;
    }

    async function callGroq(userText) {
      var key = getGroqKey();
      if (!key) {
        document.getElementById("ai-key-bar").style.display = "block";
        return "Please set your Groq API key (gear icon). Free key: console.groq.com/keys";
      }

      var shouldEdit = wantsCodeEdit(userText);
      var currentCode = cm ? cm.getValue() : "";
      var tab = getActiveTab();
      var tabName = tab ? tab.name : "Script";

      var system = [
        "You are the AI helper inside a web-based Luau code editor (Studio-style).",
        "You ARE connected to the editor. The host app will apply your code when the user confirms.",
        "NEVER say you cannot access, open, or modify the editor or Roblox Studio.",
        "NEVER tell the user to copy-paste into Roblox Studio unless they ask about Studio itself.",
        "You write Luau for Roblox (game:GetService, task.wait, Instance.new, Players, etc.).",
        ""
      ].join(" ");
      if (shouldEdit) {
        system += " The user wants you to change the open script. You MUST output the COMPLETE updated script inside one fenced code block starting with ```lua and ending with ```. Put a one-line note before the block if needed. Do not omit parts of the file. Do not refuse.";
      } else {
        system += " For normal questions, explain briefly and show small code snippets only. Do not dump a full file rewrite unless they ask to edit/fix/write the script.";
      }

      var userContent = userText;
      // Always include open script so the model can edit/reference it
      userContent += "\n\n--- Currently open script in the editor (" + tabName + ") ---\n```lua\n" + currentCode + "\n```";
      if (shouldEdit) {
        userContent += "\n\n(Instruction: return the full updated script in a ```lua code block so the editor can apply it.)";
      }
      if (aiAttachedFile) {
        userContent += "\n\n--- Attached file: " + aiAttachedFile.name + " ---\n```\n" + aiAttachedFile.content.slice(0, 12000) + "\n```";
      }

      var messages = [{ role: "system", content: system }];
      aiChatHistory.slice(-6).forEach(function(m) { messages.push(m); });
      messages.push({ role: "user", content: userContent });

      var res = await fetch("https://api.groq.com/openai/v1/chat/completions", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "Authorization": "Bearer " + key
        },
        body: JSON.stringify({
          model: "llama-3.3-70b-versatile",
          messages: messages,
          temperature: 0.4,
          max_tokens: 4096
        })
      });

      if (!res.ok) {
        var errText = await res.text();
        if (res.status === 401) return "Invalid API key. Open the gear and paste a valid Groq key.";
        if (res.status === 429) return "Rate limit hit. Wait a moment and try again.";
        return "Groq error (" + res.status + "): " + errText.slice(0, 300);
      }

      var data = await res.json();
      var reply = (data.choices && data.choices[0] && data.choices[0].message && data.choices[0].message.content) || "(empty reply)";
      aiChatHistory.push({ role: "user", content: userText });
      aiChatHistory.push({ role: "assistant", content: reply });
      return { text: reply, shouldEdit: shouldEdit };
    }

    async function sendAI() {
      var input = document.getElementById("ai-input");
      var text = input.value.trim();
      if (!text) return;

      var displayUser = escapeHtml(text);
      if (aiAttachedFile) displayUser += "<br><span style='opacity:0.7;font-size:11px;'>📎 " + escapeHtml(aiAttachedFile.name) + "</span>";
      appendAiMsg("user", displayUser);
      input.value = "";

      var thinking = appendAiMsg("bot", "<span class='ai-thinking'>● ● ●</span> Thinking…");

      try {
        var result = await callGroq(text);
        thinking.remove();
        if (typeof result === "string") {
          appendAiMsg("bot", escapeHtml(result).replace(/\n/g, "<br>"));
          return;
        }
        var el = appendAiMsg("bot", formatAiHtml(result.text));
        var code = extractCodeBlock(result.text);
        var showApply = code && cm && (result.shouldEdit || (code.split("\n").length >= 3));
        if (showApply) {
          var wrap = document.createElement("div");
          wrap.style.cssText = "margin-top:10px;display:flex;gap:8px;align-items:center;flex-wrap:wrap;";
          var btn = document.createElement("button");
          btn.className = "ai-apply-btn";
          btn.type = "button";
          btn.innerHTML = '<span style="margin-right:6px;">⬇</span> Apply to open script';
          btn.addEventListener("click", function(ev) {
            ev.preventDefault();
            ev.stopPropagation();
            applyCodeToEditor(code, btn);
          });
          wrap.appendChild(btn);
          el.appendChild(wrap);
        } else if (result.shouldEdit && !code) {
          var note = document.createElement("div");
          note.style.cssText = "margin-top:8px;font-size:12px;opacity:0.75;";
          note.textContent = 'No code block found in the reply — ask again to return the full script in a lua code block.';
          el.appendChild(note);
        }
      } catch (err) {
        thinking.remove();
        appendAiMsg("bot", escapeHtml("Network error: " + err.message));
      }

      aiAttachedFile = null;
      document.getElementById("ai-attach-preview").style.display = "none";
    }
